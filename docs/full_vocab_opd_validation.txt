"""Run one test process group, stopping before this host's 2-GiB memory limit."""
import os
from pathlib import Path
import signal
import subprocess
import sys
import time

threshold = 1850 * 1024**2
memory_file = Path('/sys/fs/cgroup/memory.current')
env = dict(os.environ, OMP_NUM_THREADS='1', OPENBLAS_NUM_THREADS='1', MKL_NUM_THREADS='1', NUMEXPR_NUM_THREADS='1', MALLOC_ARENA_MAX='2', PYTEST_DISABLE_PLUGIN_AUTOLOAD='1')
log_path, *command = sys.argv[1:]
peak = int(memory_file.read_text())
with open(log_path, 'w', buffering=1) as log:
    log.write(f'Host cgroup limit: {Path("/sys/fs/cgroup/memory.max").read_text().strip()} bytes; stop threshold: {threshold} bytes\n')
    process = subprocess.Popen(command, stdin=subprocess.DEVNULL, stdout=log, stderr=subprocess.STDOUT, start_new_session=True, env=env)
    stopped = False
    while process.poll() is None:
        current = int(memory_file.read_text())
        peak = max(peak, current)
        if current >= threshold:
            stopped = True
            os.killpg(process.pid, signal.SIGTERM)
            try:
                process.wait(timeout=2)
            except subprocess.TimeoutExpired:
                os.killpg(process.pid, signal.SIGKILL)
                process.wait()
            break
        time.sleep(0.02)
    code = process.wait()
    log.write(f'\nMemory guard: peak total cgroup memory={peak / 1024**2:.1f} MiB, threshold={threshold / 1024**2:.0f} MiB, stopped={stopped}, returncode={code}\n')
print(f'{log_path}: exit={code}, peak={peak / 1024**2:.1f} MiB, stopped={stopped}')
sys.exit(99 if stopped else code)
